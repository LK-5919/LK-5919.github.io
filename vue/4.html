# Vue3 - 自定义按钮权限指令

# 逻辑实现

directives\permission.js 文件

function isAuth(el, binding) {
    const userStore = useUserStore();
    const value = binding.value;
    // 获取当前用户所有的按钮权限
    const permission = ['user-list-add'];
    if (value && (typeof value === 'string')) {
       if(!permission.includes(value)){
        el.parentNode?.removeChild(el);
       }
    }
    else if(value && value instanceof Array && value.length > 0){
        const permissionFlag = value;
        const hasPermission = permission.some(item => {
            return permissionFlag.includes(item);
        });
        if(!hasPermission){
            el.parentNode && el.parentNode.removeChild(el);
        }
    }else{
        throw new Error(`请设置操作权限`)
    }
}

const permissionDirective = {
    mounted(el, binding) {
        isAuth(el, binding);
    },
    updated(el, binding) {
        isAuth(el, binding);
    }
};

export const setupPermissionDirective = (app) => {
    app.directive("hasPermi", permissionDirective);
};

export default permissionDirective;

\directives\index.js 文件

export * from "./permission";

# 全局注册指令

main.js 文件：

import { setupPermissionDirective } from "./directives";
import { createApp } from 'vue';
import App from './App.vue';
const app = createApp(App);
// 注册按钮指令
setupPermissionDirective(app);
app.mount('#app');

# 使用

<el-button type="primary" v-hasPermi="'agent:list:add'">添加</el-button>
