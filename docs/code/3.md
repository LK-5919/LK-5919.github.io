# WebSocket 通用配置

## 基础配置

`websocket.js`

```javascript
/**
 * @description websocket连接
 * @param {Object} options
 * @param {String} options.url websocket地址
 * @param {String?} options.onOpenEventName 连接成功时派发的自定义事件名称
 * @param {String?} options.onMessageEventName 接收到消息时派发的自定义事件名称
 * @param {String?} options.pingInterval 发送心跳的时间间隔
 * @param {Function?} options.getPingInfoFn 发送心跳的自定义数据函数
 * @param {Number?} options.reconnectCount 自动重连次数
 * @param {Number?} options.count 当前重试次数
 * @param {Boolean?} options.isAutoPing 是否发送心跳
 * @returns {Object} socket
 */
export function createSocket({
  url,
  onOpenEventName = 'onWSOpen',
  onMessageEventName = 'onMessageWS',
  pingInterval = 10000,
  getPingInfoFn = () => ({}),
  reconnectCount = 5,
  count = 0,
  isAutoPing = true
} = {}) {
  let Socket = null;
  let setIntervalWebsocketPush = null;
  let isCloseByMySelf = false;
  let reconnecting = false;
  if (typeof WebSocket === 'undefined') {
    alert('您的浏览器不支持socket');
    return null;
  }
  Socket = new WebSocket(url);
  Socket.onopen = onOpenWS;
  Socket.onmessage = onMessageWS;
  Socket.onclose = onCloseWS;
  Socket.onerror = onErrorWS;

  function onOpenWS() {
    count = 0;
    window.dispatchEvent(
      new CustomEvent(onOpenEventName, {
        detail: {
          isOpen: true,
          status: '连接成功！',
        },
      }),
    );
    isAutoPing && sendPing();
  }

  function onErrorWS() {
    Socket && Socket.close();
    clearInterval(setIntervalWebsocketPush);
    window.dispatchEvent(
      new CustomEvent(onOpenEventName, {
        detail: {
          isOpen: false,
          status: '连接失败！',
        },
      }),
    );
    if (!isCloseByMySelf && count < reconnectCount && !reconnecting) {
      reconnecting = true;
      Socket = null;
      setTimeout(() => {
        createSocket({
          url,
          onOpenEventName,
          onMessageEventName,
          pingInterval,
          getPingInfoFn,
          reconnectCount,
          count: count + 1,
          isAutoPing
        });
      }, 1000);
    }
  }

  function onMessageWS(e) {
    window.dispatchEvent(
      new CustomEvent(onMessageEventName, {
        detail: {
          data: e.data,
          type: '这是来自服务端的数据',
        },
      }),
    );
  }

  function connecting(message) {
    setTimeout(() => {
      if (Socket && Socket.readyState === 0) {
        connecting(message);
      } else {
        Socket.send(JSON.stringify(message));
      }
    }, 1000);
  }

  function sendWSPush(message) {
    if (Socket && Socket.readyState === 1) {
      Socket.send(JSON.stringify(message));
    } else if (Socket && Socket.readyState === 0) {
      connecting(message);
    }
  }

  function onCloseWS() {
    clearInterval(setIntervalWebsocketPush);
    if (!isCloseByMySelf && count < reconnectCount && !reconnecting) {
      reconnecting = true;
      Socket = null;
      setTimeout(() => {
        createSocket({
          url,
          onOpenEventName,
          onMessageEventName,
          pingInterval,
          getPingInfoFn,
          reconnectCount,
          count: count + 1,
          isAutoPing
        });
      }, 1000);
    }
  }

  function sendPing() {
    setIntervalWebsocketPush && clearInterval(setIntervalWebsocketPush);
    Socket.send(JSON.stringify(getPingInfoFn()));
    setIntervalWebsocketPush = setInterval(() => {
      Socket.send(JSON.stringify(getPingInfoFn()));
    }, pingInterval);
  }

  function closeWebsocket() {
    isCloseByMySelf = true;
    Socket && Socket.close();
    Socket = null;
  }

  return {
    sendWSPush,
    closeWebsocket,
  };
}
```

## 使用

在 pinia 中使用

```javascript
import { defineStore } from 'pinia';
import { createSocket } from '@/utils/websocket.js';

const useWebsocketStore = defineStore('websocket', {
  state: () => ({
    socket: null,
    isOpen: false,
    hasData: false,
  }),
  actions: {
    create({
      url = 'ws://localhost:9991',
      onOpenEventName = 'onOpen',
      onMessageEventName = 'onMessageWS',
      pingInterval = 10000,
      reconnectCount = 5,
      count = 0,
    } = {}) {
      if (!this.isOpen) {
        this.socket = createSocket({
          url,
          onOpenEventName,
          onMessageEventName,
          pingInterval,
          getPingInfoFn,
          reconnectCount,
          count,
        });
      }
    },
    send(data) {
      if (this.isOpen) {
        this.socket.sendWSPush({
          code: '02',
          data: [data],
        });
      }
    },
    close() {
      this.socket.closeWebsocket();
    },
  },
});

function getPingInfoFn() {
  let ping = {
    code: '03',
    data: null,
  };
  return ping;
}
export default useWebsocketStore;
```
