# jsx 基础

## 插值语法

在 jsx 中，使用 `{}` 符号来进行插值表达

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    const msg=ref('hello')
    return ()=><div> {msg.value} </div>
  }
})
```

## 条件渲染 v-if

在 jsx 中不存在 `v-if` 指令，需要使用 JavaScript 原生提供的 if 判断或者三元运算符来表示

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    const msg=ref('zerotower')
    const body=<div>{msg.value}</div>
    const visible=ref(false);
    return ()=><div> {visible.value? body:<></>} </div>
  }
})
```

## 条件渲染 v-show

在 jsx 中，可以使用 `v-show` 指令

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    const msg=ref('hello');
    const canshow=ref(true)
    return ()=><div v-show={!canshow.value}> zerotower </div>
  }
})
```

## 列表循环

在 jsx 中，不存在 `v-for` 指令，需要使用 JavaScript 原生的循环函数来进行遍历，需要注意的是，在 jsx 的循环中，仍需要指定 `key`

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    const nameList=ref(['David','Susan','Jack'])
    return ()=><div> {
      nameList.value.map((name,index)=>(<div key={index}>name:{name}</div>))
    } </div>
  }
})
```

## 事件绑定

在 jsx 中，不在使用 `@` 符号来指定事件，采用类似 `onClick` 的写法进行事件绑定

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
 name:"app",
 setup(props,ctx){
   const count=ref(0);
   function handleCount(){
     count.value++;
   }
   return ()=><div>
   <div>{count.value}</div>
   <button onClick={handleCount}>+</button>
   </div>
  }
})
```

参数传递

```javascript
function handleCount(step){
      count.value+=step;
    }
    return ()=><div>
      <div>{count.value}</div>
      <button onClick={()=>handleCount(3)}>+</button>
      </div>
  }
```

## 事件修饰符

在 jsx 中，借助 `withModifiers` 来进行事件修饰

```javascript
import {defineComponent, ref,withModifiers} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    return ()=><div>
      <div onClick={withModifiers(()=>{console.log('我被命中了')},["self"])}><span>zerotower</span></div>
      </div>
  }
})
```

但是， 对于修饰符 `.passive`、`.capture`、`.once`，`withModifiers` 并不生效，我们需要使用链式驼峰的写法，以下是使用 `.once` 的一个例子：

```javascript
import {defineComponent, ref,withModifiers} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
    return ()=><div>
      <div onClickOnce={()=>{console.log('我被命中了')}}><span>zerotower</span></div>
      </div>
  }
})
```

## V-model 数据绑定

在 jsx 中是可以直接使用 v-model 指令的

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"app",
  setup(props,ctx){
   const text=ref("")
    return ()=><div>
      <input type="text" v-model={text.value}/>
      </div>
  }
})
```

## 插槽使用

- 普通插槽与具名插槽

jsx 子组件

```javascript
import {defineComponent, ref} from "vue"
export default defineComponent({
  name:"Children ",
  setup(props, { slots }){
     return () => (
      <div>
        {/* 默认插槽 */}
        {slots.default?.()}
        
        {/* 具名插槽 */}
        {slots.namedSlot && <div>{slots.namedSlot()}</div>}
      </div>
    );
  }
})
```

父组件使用

```javascript
<template>
    <div>
        <Children>
            <!-- 默认插槽内容 -->
            <template #default>
              <p>这是默认插槽的内容</p>
            </template>
            <!-- 具名插槽内容 -->
            <template #namedSlot>
              <p>这是具名插槽的内容</p>
            </template>
        </Children>
    </div>
</template>
<script setup>
import Children from "./children.jsx";
</script>
```

- 作用域插槽

jsx 子组件

```javascript
import { defineComponent, h } from 'vue';

export default defineComponent({
  name: 'MyComponent',
  props: {
    // ... 你的props定义
  },
  setup(props, { slots }) {
    // 假设有一个带作用域数据的插槽
    const scopedSlotData = { message: 'Hello from the child component!' };

    return () => (
      <div>
        {/* 使用函数来接收并渲染作用域插槽 */}
        {slots.myScopedSlot && (
          <div>
            {slots.myScopedSlot(scopedSlotData)}
          </div>
        )}
      </div>
    );
  }
});
```

父组件使用

```javascript
<template>
  <MyComponent>
    <!-- 定义作用域插槽 -->
    <template #myScopedSlot="slotProps">
      <p>The message is: {{ slotProps.message }}</p>
    </template>
  </MyComponent>
</template>
<script setup>
import MyComponent from "./MyComponent.jsx";
</script>
```

## 编写 jsx 函数组件

```javascript
// 模拟实现一个messageBox
import './messageBox.css'; // 将scss文件转换成css文件后导入
import { render } from 'vue';
export function showMessageBox(content, handler) {
  const div = document.createElement('div');
  const box = <div class='box-cover'>
    <div class='box'>
        <div class='content'>
            <div class='title'>提示</div>
            <div class='main'>
                <div class='text'>{content}</div>
            </div>
            <div class='btns'>
                <button class='cancel-btn' onClick={() => {
                    document.body.removeChild(div);
                    handler.cancel && handler.cancel();
                }}>取消</button>
                <button class='confirm-btn' onClick={() => {
                    document.body.removeChild(div);
                    handler.confirm && handler.confirm();
                }}>确定</button>
            </div>
        </div>
    </div>
  </div>
  render(box, div);
  document.body.appendChild(div);
}
```

- 使用此函数式组件

```javascript
import { showMessageBox } from './components/messageBox'; 
showMessageBox('测试弹窗', {
    confirm: () => {
      console.log('点击了确定');
    },
    cancel: () => {
      console.log('点击了取消');
    }
  })
```
