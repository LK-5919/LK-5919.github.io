# Vue3 - 路由守卫

## 全局 store 中处理动态路由

`\store\modules\user.js` 文件, 使用 `Pinia` 作为状态管理器

```javascript
import {
  login,
  logout,
  getUserInfo,
  getAgentLogo
} from '@/api/login';

import {
  getToken,
  setToken,
  removeToken
} from '@/utils/auth';

// 引入头像
import avatar from '@/assets/images/user-avatar.png';

// 匹配views里面所有的.vue文件
const modules = import.meta.glob('@/views/**/*.vue');
export const loadView = (view) => {
  let res;
  for (const path in modules) {
    const dir = path.split('views/')[1].split('.vue')[0];
    if (dir === view) {
      res = () => modules[path]();
    }
  }
  return res;
}

const useUserStore = defineStore(
  'user', {
  state: () => ({
    token: getToken(),
    userInfo: {},
    menuList: [],
    permission: [],
    dynamicRoutes: [],
    isEmptyRoutes: false,
    avatar,//用户头像
  }),
  actions: {
    // 登录
    login(userInfo) {
      return new Promise((resolve, reject) => {
        login(userInfo).then(res => {
          setToken(res.data.cache_user_access_token.access_token);
          this.token = res.data.cache_user_access_token.access_token;
          resolve(res);
        }).catch(error => {
          reject(error)
        })
      })
    },
    // 获取用户信息
    getInfo() {
      return new Promise((resolve, reject) => {
        getUserInfo().then(res => {
          this.userInfo = res.data.user_do;
          this.permission = res.data.user_menus.permissions;
          if (res.data.user_menus.sider && res.data.user_menus.sider.length > 0) {
            this.menuList = res.data.user_menus.sider;
            this.dynamicRoutes = PatterMenu(res.data.user_menus.sider);
            resolve(this.dynamicRoutes);
          } else {
            this.isEmptyRoutes = true;
            resolve([]);
          }
        }).catch(error => {
          this.isEmptyRoutes = false;
          reject(error)
        })
      })
    },
    // 退出系统
    logOut() {
      return new Promise((resolve, reject) => {
        logout().then(() => {
          removeToken();
          this.token = ''
          this.userInfo = {}
          this.menuList = []
          this.permission = []
          this.dynamicRoutes = []
          this.isEmptyRoutes = false;
          resolve()
        }).catch(error => {
          reject(error)
        })
      })
    }

  }
})

// 解析路由数据--获取没有子集的路由
function PatterMenu(list) {
  let arr = [];
  // 递归解析路由表
  function routesArr(args) {
    args && args.length && args.forEach((item) => {
      if (item.menu_type === '1' && item.children && item.children.length > 0) {
        routesArr(item.children);
      }
      if (item.menu_type === '2' && item.children === null) {
        arr.push({
          path: `${item.menu_path}`,
          component: loadView(item.component),
          meta: { title: useI18n.t(item.international_code) },
          name: useI18n.t(item.international_code),
        });
      }
    });
  }
  // 默认调用
  routesArr(list);
  return arr;
}
export default useUserStore;
```

## 守卫中添加动态路由

`\router\guard\permission.js` 文件

```javascript
import useUserStore from '@/store/modules/user'；
import {
  getToken
} from '@/utils/auth'；
const whiteList = ['/login']；
// 前置路由守卫
export const createPermission = (router) => {
  router.beforeEach((to, from, next) => {
    if (getToken()) {
      // 存在token
      if (to.path === '/login') {
        router.replace({ path: "/" });
      }
      if (useUserStore().menuList.length === 0) {
        useUserStore()
          .getInfo()
          .then((list) => {
            if (list.length > 0 && !useUserStore().isEmptyRoutes) {
              list.forEach(item => {
              // 动态添加路由
                router.addRoute("Layout", item)
              })
              next({
                ...to,
                replace: true
              });
            } else {
              next()
            }
          })
          .catch((err) => {
            if (err.code === 423406 || err.code === 423405) {
              // 拉取用户权限菜单token过期时，直接强制刷新页面到登录页
              location.reload();
            } else {
            }
          })
      } else {
        next();
      }
    } else {
      if (whiteList.indexOf(to.path) !== -1) {
        next();
        return;
      }
      next(`/login?redirect=${to.fullPath}`) // 否则全部重定向到登录页
    }
  })
}
```

`\router\guard\index.js` 文件

```javascript
import { createPermission } from "./permission";

export function setupRouterGuard(router) {
    createPermission(router);
}
```

`\router\index.js` 文件

```javascript
import {
  createWebHistory,
  createRouter,
  createWebHashHistory,
} from 'vue-router'

// 引入layout
import Layout from '@/layout'

// 公共路由
export const constantRoutes = [
  {
    path: '/redirect',
    component: Layout,
    hidden: true,
    children: [
      {
        path: '/redirect/:path(.*)',
        component: () => import('@/views/redirect/index.vue'),
      },
    ],
  },
  {
    path: '/login',
    component: () => import('@/views/login'),
    hidden: true,
  },
  {
    path: '/:pathMatch(.*)*',
    component: () => import('@/views/error/404'),
    hidden: true,
  },
  {
    path: '/401',
    component: () => import('@/views/error/401'),
    hidden: true,
  },
  {
    path: '',
    component: Layout,
    redirect: '/index',
    name: 'Layout',
    children: [
      {
        path: '/index',
        component: () => import('@/views/agent'),
        name: 'index',
        meta: { title: 'message.loginText' },
      },
    ],
  },

]

const router = createRouter({
  history: createWebHashHistory(),
  routes: constantRoutes,
  scrollBehavior(to, from, savedPosition) {
    if (savedPosition) {
      return savedPosition
    } else {
      return { top: 0 }
    }
  },
})

export default router;
```

## 全局注册路由守卫

`main.js` 文件

```javascript
import { createApp } from 'vue';
import App from './App.vue';
import router from './router';
import { setupRouterGuard } from "@/router/guard";
const app = createApp(App);
// 注册路由守卫
setupRouterGuard(router);
app.mount('#app');
```
