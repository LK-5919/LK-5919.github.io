# Vue3 - 图片懒加载指令

使用 `IntersectionObserver` API 实现。

## 逻辑实现

`src/directives/index.js`

```javascript
**export** * **from** '[./lazyLoad.js](https://gitee.com/guangzhou-zhongke-kaize_1/sportsbot-h5/blob/master/src/directives/lazyLoad.js)';
```

`src/directives/lazyLoad.js`

```javascript
**const** lazyLoadDirective = {
  mounted(el, bind) {
    **if** (bind.arg) {
      **let** options = {
        root: **null**,
        rootMargin: '0px',
        threshold: **0.1**,
      };
      **let** observer = **new** IntersectionObserver((entries, observer) => {
        **if** (entries[0].intersectionRatio >= **0.1**) {
          **let** imgUrl = bind.value;
          **if** (bind.value === '' || bind.value === **undefined**) {
            imgUrl = '/public/image/pic.svg';
          }
          entries[0].target.firstChild.src = imgUrl;
          // 替换后移除dom监视
          observer.unobserve(el);
        }
      }, options);
      observer.observe(el);
    }
  },
};
**export** **const** setuplazyLoadDirective = (app) => {
  app.directive('lazyLoad', lazyLoadDirective);
};
```

## 全局命令注册

`main.js`

```javascript
**import** { createApp } **from** '[vue](https://github.com/vuejs/core)';
**import** { setuplazyLoadDirective } **from** '@/directives/index.js';
**import** App **from** '[./App.vue](https://gitee.com/guangzhou-zhongke-kaize_1/sportsbot-h5/blob/master/src/App.vue)';
**const** app = createApp(App);
**const** setupDirective = () => {
  // 注册lazyLoad自定义指令
  setuplazyLoadDirective(app);
};
setupDirective();
app.mount('#app');
```

## 使用

```javascript
<**template**>
  <div class="container">
   _<!-- 图片组件 -->_
        <van-image
          :src="isOpen ? '/image/pic.svg' : item.imageUrl"
          fit="cover"
          v-lazyLoad:[isOpen]="item.imageUrl"
        >
          <template v-slot:loading>
            <van-loading type="spinner" size="20" />
          </**template**>
          <**template** v-slot:error>加载失败</**template**>
        </van-image>
  </div>
</template>
<**script** setup>
**const** props = defineProps({
  // 是否开启图片懒加载 -- 默认开启
  isOpen: {
    type: Boolean,
    default: **true**,
  },
  item: {
    type: Object,
    required: **true**,
  },
});
</**script**>
<**style** lang="less" scoped></style>
```
