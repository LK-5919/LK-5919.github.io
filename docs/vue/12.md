## 逻辑实现

1. `src/index.js`

```javascript
**#!/usr/bin/env node**
**import** { program } **from** '[commander](https://github.com/tj/commander.js)';
**import** fs **from** 'node:fs';
**import** path **from** 'node:path';
**import** { fileURLToPath } **from** 'node:url';
**import** inquirer **from** '[inquirer](https://github.com/SBoudrias/Inquirer.js)';
**import** { checkPath, downloadTemp } **from** '[./utils.js](https://gitee.com/LkGiteeCoder/my-vue-cli/blob/main/src/utils.js)';
**const** currentDir = path.dirname(fileURLToPath(**import**.meta.url));
**let** json = fs.readFileSync(path.join(currentDir, "../",'package.json'), 'utf-8');
json = JSON.parse(json);
program.version(json.version);
program.command('create <projectName>').alias('c').description('create a new project').action((projectName)=>{
    inquirer.prompt([{
        type: 'input',
        name: 'projectName',
        message: 'please enter project name',
        default: projectName
    },
    {
        type: 'confirm',
        name: 'isTs',
        message: 'support TypeScript',
    },
    {
        type: 'confirm',
        name: 'isH5',
        message: 'is mobile project',
    },
]).then(answer => {
        **if**(checkPath(answer.projectName)){
            console.log('projectName is exist!');
            **return**;
        }**else**{
            **let** branch = '';
            answer.isTs ? (branch = 'ts') : (branch = 'js');
            answer.isH5 ? (branch = branch + '-h5') : **null**;
            downloadTemp(branch, answer.projectName);
        }
    })
});
program.parse(process.argv)
```

1. `src/utils.js`

```javascript
**#!/usr/bin/env node**
**import** fs **from** 'node:fs';
**import** download **from** '[download-git-repo](https://gitlab.com/flippidippi/download-git-repo.git)';
**import** ora **from** '[ora](https://github.com/sindresorhus/ora)';
**const** spinner = ora('download...')
**export** **const** checkPath = (path) => {
    **if**(fs.existsSync(path)){
        **return** **true**;
    }**else**{
        **return** **false**;
    }
}

**export** **const** downloadTemp = (branch, project) => {
    spinner.start();
    **return** **new** Promise((resolve, reject) => {
        // 填自己的项目仓库地址
        download(`direct:git@gitee.com:LkGiteeCoder/vue-template.git#${branch}`, project, { clone: **true**},(e)=>{
            **if**(e){
                console.log('download error: ',e);
               reject(e); 
            }
            resolve();
            spinner.succeed('download finish');
        })
    })
}
```

## 使用

上述文件实现后，在此项目根目录下启动终端，使用 npm link 命令将此项目配置的 bin 下的 'lk-cli' 挂载到全局下即可；
