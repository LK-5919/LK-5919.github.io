# Vue3 - swiper

## 安装依赖

```javascript
pnpm i swiper
```

## 组件中使用

```html
<template>
        <div class="type-wrap">
          <swiper
            :loop="false"
            :space-between="30"
            :modules="modules"
            :pagination="pagination"
            :slidesPerView="'auto'"
            :observer="true"
            :observeParents="true"
            class="swiperBox"
          >
            <swiper-slide class="slide" v-for="item in list" :key="item.id">
              <div class="item">
              </div>
            </swiper-slide>
          </swiper>
        </div>
</template>
<script setup>
import { Swiper, SwiperSlide } from 'swiper/vue';
import { Pagination } from 'swiper/modules';
import 'swiper/css';
import 'swiper/css/pagination';
import { ref, watchEffect } from 'vue';
const modules = [Pagination];
const pagination = {
  clickable: true,
  renderBullet: function (index, className) {
    return '<span class="' + className + '"></span>';
  },
};
const list = ref([]);
watchEffect(() => {
 list.value = [
      {
        id: 1,
        desc: '1',
      },
      {
        id: 2,
        desc: '2',
      },
      {
        id: 3,
        desc: '3',
      },
    ];
});
</script>
<style lang="less" scoped>
  .type-wrap {
      .swiperBox {
        width: 100%;
        height: 340px; // 指定高度

        .slide {
          width: 400px;
          display: flex; 
          justify-content: center;
          align-items: center;

          .item {}
        }
      }
    }
</style>
```
