# Vue3 - Vue-i18n 配置

## 基础配置

`./translate/index.js`:

```javascript
import zh from './zh.js';
import en from './en.js';
const translate = {
  zh,
  en,
};
export default translate;
```

`/lang/index.js`:

```javascript
import { createI18n } from 'vue-i18n'
import translate from './translate';
import axios from 'axios'

// 请求获取语言文件
const getLangJS = (lang) => {
    return new Promise((resolve, reject) => {
        axios({
            url: `${import.meta.env.VITE_APP_BASE_LANG_URL}/translate/${lang}.js`,
            responseType: 'arraybuffer'
        }).then(res => {
            const arrayBuffer = res.data;
            const blob = new Blob([arrayBuffer], {
                type: 'application/javascript'
            });
            const blobURL = URL.createObjectURL(blob);
            import(/* @vite-ignore */blobURL).then(module => {
                const context = module.default;
                resolve({ lang, context })
            }).catch(e => {
                // console.log("打印 --> fail add lang", e);
                reject(e)
            })
        }).catch(err => {
            reject(err)
        })
    })

}

// 创建并配置i18n
const i18n = createI18n({
    legacy: false,  // 没有该参数可能会报错
    locale: sessionStorage.getItem('language_type') ? sessionStorage.getItem('language_type') : 'zh', //zh是中文，en是英文
    messages: {
        zh: translate.zh,
        en: translate.en
    }
})


/*
 在vuei18n框架中有两个方法，可以把后台接口拉取到的数据合并到框架内部的语言环境中
1.  setLocaleMessage( locale, message )
    参数：
      {Locale} locale
      {LocaleMessageObject} message
  设置语言环境的 locale 信息
 
2.  mergeLocaleMessage( locale, message )
    参数：
      {Locale} locale
      {LocaleMessageObject} message
  将语言环境信息 locale 合并到已注册的语言环境信息中。
 
两种使用方法:
      setLocaleMessage会重设框架的语言环境，如果在调用这个方法之前语言环境里面有文件那么会覆盖这个文件，
      mergeLocaleMessage不会覆盖掉他只会合并两个文件，
      所以当你既要做网站的多语言，又需要element框架的国际化时，就需要用mergeLocaleMessage。
 */


// 添加请求语言到i18n中
const addLangJS = async () => {
    let promiseList = []
    let locale = sessionStorage.getItem('language_type') ? sessionStorage.getItem('language_type') : 'zh'
    promiseList.push(getLangJS(locale));
    let flag = false
    await Promise.all(promiseList).then(res => {
        flag = true
        res.map(v => {
            i18n.global.setLocaleMessage(v.lang, v.context);
        })
    }).catch(err => {
        flag = false
    })
    return flag
}

// 请求获取翻译文件 -- 在动态获取语言文件时使用
if (!await addLangJS()) {
    if (i18n.global.locale.value === 'zh') {
        ElMessage.error('语言文件配置错误')
    } else {
        ElMessage.error('LANGUAGE FILE CONFIGURATION ERROR!')
    }
}
// 在script文件中使用useI8n方法进行翻译
export const useI18n = i18n.global;
export default i18n;
```

## 使用

`main.js` 文件注册

```javascript
// 国际化组件
import i18n from '@/language/i18n';
import { createApp } from 'vue';
import App from './App.vue';
const app = createApp(App);
app.use(i18n);
app.mount('#app');
```

在 `template` 中使用:

```html
<div>
      {{ $t(`staffAttendanceDetail.sureExportAllData`) }}
 </div>
```

在 `script` 文件中使用：

```javascript
import { useI18n } from '@/lang/index.js';
const msg = useI18n.t('test.msg');
```
